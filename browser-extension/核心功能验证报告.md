**代码位置**：[content.js](file:///c:\MsSafeData\Desktop\yijiandaodi\browser-extension\content\content.js)

**实现原理**：
```javascript
// 平台配置（38-87行）
const PLATFORMS = {
  DEEPSEEK: {
    name: 'DeepSeek',
    domain: 'chat.deepseek.com',
    type: 'ai_chat',
    contentSelector: '.message-content'
  },
  CHATGPT: {...},
  ERNIE: {...},
  ...
}

// 平台检测函数（151行）
function detectCurrentPlatform() {
  // 缓存机制，避免重复检测
  if (detectedPlatformCache) {
    return detectedPlatformCache;
  }

  // 基于域名匹配
  const hostname = window.location.hostname;

  for (const key of Object.keys(PLATFORMS)) {
    const platform = PLATFORMS[key];
    if (hostname.includes(platform.domain)) {
      detectedPlatformCache = platform;
      return platform;
    }
  }

  return null; // 未识别平台
}
```

### 3.3 内容预览限制

**原因**：
- Storage空间限制（5MB）
- 防止保存大量完整内容
- 提升性能

**策略**：
- 文本内容：只存前100字符预览
- 图片内容：只存URL，不下载文件
- 音频内容：只存URL，不下载文件
- 完整内容：在报告中仅显示预览
